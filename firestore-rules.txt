rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ---- Who is signing in ----
    function signedIn() { return request.auth != null && request.auth.token.email_verified == true; }
    function myEmail()  { return request.auth.token.email.lower(); }
    // The admin manages staff, the home page and data.
    function isAdmin()  { return signedIn() && myEmail() == 'vijayjalneela@gmail.com'; }
    function staffPath() { return /databases/$(database)/documents/staff/$(myEmail()); }
    function isStaffMember() { return signedIn() && exists(staffPath()); }
    function me()       { return get(staffPath()).data; }
    function isPrincipal() { return isAdmin() || (isStaffMember() && me().role == 'principal'); }
    function isTeacher()   { return isStaffMember() && me().role == 'teacher'; }
    function anyStaff()    { return isAdmin() || isStaffMember(); }
    // Coordinator: marked by the admin on the staff list. While she is on leave the principal can name
    // an acting coordinator in office/acting; that power switches off by itself after the 'until' time.
    function actingPath() { return /databases/$(database)/documents/office/acting; }
    function isActing() { return exists(actingPath()) && get(actingPath()).data.email == myEmail()
                            && request.time >= get(actingPath()).data.since && request.time < get(actingPath()).data.until; }
    function isCoord()  { return isTeacher() && (me().get('coordinator', false) == true || isActing()); }
    // "Office" = principal, admin or (acting) coordinator: runs students, attendance, leave, timetable, exams.
    function isOffice() { return isPrincipal() || isCoord(); }

    // ---- Helpers for one report card ----
    function cls(d) { return d.std + '-' + d.div; }
    function isClassTeacherOf(d) { return isTeacher() && cls(d) in me().get('classTeacher', []); }
    function marksChanged() {
      return request.resource.data.get('marks', {}).diff(resource.data.get('marks', {})).affectedKeys();
    }
    // A teacher may only change marks of her own subjects, in classes she teaches.
    // If the admin set "which subjects in each class" (teaches), that list decides; otherwise
    // all her subjects in all her classes.
    function hasMap() { return ('teaches' in me()) && me().teaches.size() > 0; }
    function onlyOwnSubjects() {
      return marksChanged().size() == 0
        || (hasMap() && marksChanged().hasOnly(me().teaches.get(cls(resource.data), [])))
        || (!hasMap() && cls(resource.data) in me().get('classes', [])
            && marksChanged().hasOnly(me().get('subjects', [])));
    }
    function onlyMarks() {
      return request.resource.data.diff(resource.data).affectedKeys().hasOnly(['marks', 'updatedAt', 'updatedBy']);
    }

    // Basic shape checks on every saved card (photo size, class field consistent).
    function okCard(d) {
      return d.std is string && d.div is string && d.year is string && d.term is string
        && (!('photo' in d) || d.photo == null || (d.photo is string && d.photo.size() < 150000))
        && (!('cls' in d) || d.cls == cls(d));
    }
    function teachesCard(d) { return isTeacher() && (cls(d) in me().get('classes', []) || isClassTeacherOf(d)); }

    // ---- Report cards (full records): STAFF ONLY ----
    match /results/{id} {
      allow get, list: if anyStaff();
      allow create: if (isOffice() || isClassTeacherOf(request.resource.data)) && okCard(request.resource.data);
      allow delete: if isOffice() || isClassTeacherOf(resource.data);
      allow update: if okCard(request.resource.data) && (isOffice()
        || (isTeacher() && onlyOwnSubjects()
            && (onlyMarks()
                || (isClassTeacherOf(resource.data) && cls(request.resource.data) == cls(resource.data)))));
    }

    // ---- Parent copies ----
    // The document name is a SHA-256 hash of year, exam, class, roll, name AND date of birth,
    // so a parent must know all of them. Nobody can list or browse this collection.
    match /reports/{key} {
      allow get: if true;
      allow list: if false;
      allow create, update: if (isOffice() || teachesCard(request.resource.data)) && okCard(request.resource.data);
      allow delete: if isOffice() || teachesCard(resource.data);
    }

    // ---- Staff list (roles, subjects, classes): the admin, or the principal approving the coordinator's
    //      recommendation. Only the admin can add, change or remove a principal. ----
    match /staff/{em} {
      // Anyone signed in (even before verifying) may check only their OWN entry,
      // so the site can confirm an email is on the list before sending a link.
      allow get: if (request.auth != null && em == request.auth.token.email.lower()) || anyStaff();
      allow list: if anyStaff();
      allow create: if (isAdmin() || (isPrincipal() && request.resource.data.role == 'teacher'))
        && (!('photo' in request.resource.data) || request.resource.data.photo == null || request.resource.data.photo.size() < 150000);
      allow update: if (isAdmin() || (isPrincipal() && resource.data.role == 'teacher' && request.resource.data.role == 'teacher'))
        && (!('photo' in request.resource.data) || request.resource.data.photo == null || request.resource.data.photo.size() < 150000);
      allow delete: if isAdmin() || (isPrincipal() && resource.data.role == 'teacher');
    }

    // ---- Contract details (type, renewal / re-tender and termination dates) ----
    // Private: the principal, admin and coordinator can read; only the principal / admin change them
    // (the coordinator's changes arrive as drafts for approval). Teachers cannot see them.
    match /staffHR/{em} {
      // The coordinator sees teachers' contracts only, never the principal's.
      allow read: if isPrincipal() || (isCoord() && resource.data.get('role', '') == 'teacher');
      allow write: if isPrincipal();
    }

    // ---- Timetable (one document per year): all staff can see it, principal/admin change it ----
    match /timetable/{year} {
      allow read: if anyStaff();
      allow write: if isPrincipal();
    }

    // ---- Leave: a teacher can request leave for herself and see only her own;
    //      the principal/admin see all, approve or reject. ----
    match /leaves/{id} {
      // The coordinator can only recommend; the principal approves or rejects.
      allow read: if isOffice() || (isStaffMember() && resource.data.email == myEmail());
      allow create: if isPrincipal()
        || (isCoord() && request.resource.data.status == 'Recommended')
        || (isStaffMember() && request.resource.data.email == myEmail() && request.resource.data.status == 'Requested');
      allow update: if isPrincipal()
        || (isCoord() && resource.data.status == 'Requested' && request.resource.data.status == 'Recommended');
      allow delete: if isPrincipal() || (isCoord() && resource.data.status in ['Requested', 'Recommended'])
        || (isStaffMember() && resource.data.email == myEmail() && resource.data.status == 'Requested');
    }

    // ---- School calendar (holidays, vacations, exam dates): everyone can see it ----
    match /calendar/{year} {
      allow read: if true;
      allow write: if isPrincipal();
    }

    // ---- Exam invigilation duty (one document per year): staff see it, principal/admin assign it ----
    match /duties/{year} {
      allow read: if anyStaff();
      allow write: if isPrincipal();
    }

    // ---- Substitute plans (one document per date): published by the principal, seen by staff ----
    match /subs/{date} {
      allow read: if anyStaff();
      allow write: if isPrincipal();
    }

    // ---- Drafts: the coordinator prepares the timetable, calendar, exam duty, substitutes and the
    //      disclosure page here; only the principal / admin publishes them (writes the real document). ----
    match /drafts/{key} {
      allow read: if isOffice();
      allow create, update: if isCoord() && request.resource.data.col in ['timetable', 'calendar', 'duties', 'subs', 'config', 'staff', 'staffHR', 'import', 'action']
        && key == request.resource.data.col + '__' + request.resource.data.id;
      allow delete: if isOffice();
    }

    // ---- Acting coordinator (only the principal / admin chooses) ----
    match /office/{id} {
      allow read: if anyStaff();
      allow write: if isPrincipal();
    }

    // ---- Student register: status (studying / left), section moves, parent phone. Staff only. ----
    function okStu(d) { return d.year is string && d.std is string && d.div is string; }
    match /students/{sid} {
      allow read: if anyStaff();
      allow create: if okStu(request.resource.data) && (isOffice()
        || (isClassTeacherOf(request.resource.data) && request.resource.data.get('status', 'active') == 'active'));
      allow update: if okStu(request.resource.data) && (isOffice()
        || (isClassTeacherOf(resource.data) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['phone', 'consent', 'history', 'updatedAt', 'updatedBy'])));
      allow delete: if isOffice();
    }

    // ---- Health and discipline notes: only teachers of that class, the coordinator and principal.
    //      Never copied to the parents' report cards. ----
    match /studentNotes/{sid} {
      allow read: if isOffice() || teachesCard(resource.data);
      allow create: if okStu(request.resource.data) && (isOffice() || isClassTeacherOf(request.resource.data));
      allow update: if okStu(request.resource.data) && (isOffice() || (isClassTeacherOf(resource.data) && isClassTeacherOf(request.resource.data)));
      allow delete: if isOffice();
    }

    // ---- Daily class attendance: one document per class per month, marked by the class teacher ----
    match /attendance/{id} {
      allow read: if anyStaff();
      allow create, update: if okStu(request.resource.data)
        && id == request.resource.data.year + '_' + cls(request.resource.data) + '_' + request.resource.data.month
        && (isOffice() || isClassTeacherOf(request.resource.data));
      allow delete: if isOffice();
    }

    // ---- TC check: public can only open one record by its hash (TC no. + name); no browsing ----
    match /tcs/{key} {
      allow get: if true;
      allow list: if false;
      allow create, update: if isOffice() && request.resource.data.keys().hasOnly(['tc', 'name', 'cls', 'issued', 'sample'])
        && request.resource.data.tc is string && request.resource.data.name is string;
      allow delete: if isOffice();
    }

    // ---- Staff attendance: principal and coordinator only ----
    match /staffAtt/{id} {
      allow read, write: if isOffice();
    }

    // ---- Home page: school photo and principal's message (principal or admin) ----
    match /config/{id} {
      allow read: if true;
      allow write: if isPrincipal();
    }
  }
}
